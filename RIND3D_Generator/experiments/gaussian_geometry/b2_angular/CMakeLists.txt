cmake_minimum_required(VERSION 3.23)
project(rind3d_b2_angular LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_library(GMP_LIBRARY gmp REQUIRED)
find_library(GMPXX_LIBRARY gmpxx REQUIRED)
find_library(MPFR_LIBRARY mpfr REQUIRED)
pybind11_add_module(_b2_angular bindings.cpp geometry.cpp field.cpp ../b2_native/primary.cpp ../b2_native/reference.cpp)
target_include_directories(_b2_angular PRIVATE ../b2_native)
target_link_libraries(_b2_angular PRIVATE ${MPFR_LIBRARY} ${GMPXX_LIBRARY} ${GMP_LIBRARY})
target_compile_options(_b2_angular PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
