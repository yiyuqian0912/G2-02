cmake_minimum_required(VERSION 3.23)
project(rind3d_shadow_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_library(GMP_LIBRARY gmp REQUIRED)
find_library(GMPXX_LIBRARY gmpxx REQUIRED)
pybind11_add_module(_shadow_native native.cpp ../../native/filter.cpp)
target_link_libraries(_shadow_native PRIVATE ${GMPXX_LIBRARY} ${GMP_LIBRARY})
target_compile_options(_shadow_native PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
