cmake_minimum_required(VERSION 3.23)
project(rind3d LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_library(ZSTD_LIBRARY zstd REQUIRED)
set(RINDRAY_PYTHON OFF CACHE BOOL "" FORCE)
set(RINDRAY_VULKAN OFF CACHE BOOL "" FORCE)
set(RINDRAY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RINDRAY_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/RINDRay" CACHE PATH "Existing RINDRay checkout")
if(NOT EXISTS "${RINDRAY_SOURCE}/include/rindray/rindray.hpp")
  message(FATAL_ERROR "Set RINDRAY_SOURCE to the existing RINDRay checkout")
endif()
add_subdirectory("${RINDRAY_SOURCE}" rindray-core EXCLUDE_FROM_ALL)
pybind11_add_module(_native native/module.cpp native/pack.cpp native/filter.cpp native/functional.cpp)
target_link_libraries(_native PRIVATE rindray ${ZSTD_LIBRARY})
target_compile_options(_native PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
install(TARGETS _native LIBRARY DESTINATION rind3d)
find_library(GMP_LIBRARY gmp REQUIRED)
find_library(GMPXX_LIBRARY gmpxx REQUIRED)
pybind11_add_module(_convex native/convex.cpp)
target_link_libraries(_convex PRIVATE ${GMPXX_LIBRARY} ${GMP_LIBRARY})
target_compile_options(_convex PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
install(TARGETS _convex LIBRARY DESTINATION rind3d)
pybind11_add_module(_research native/research.cpp native/pack.cpp)
target_link_libraries(_research PRIVATE ${GMPXX_LIBRARY} ${GMP_LIBRARY} ${ZSTD_LIBRARY})
target_compile_options(_research PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
install(TARGETS _research LIBRARY DESTINATION rind3d)
